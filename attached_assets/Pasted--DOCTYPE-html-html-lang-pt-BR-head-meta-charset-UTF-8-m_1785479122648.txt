<script>
  var layout  = document.getElementById('st-layout');
  var backBtn = document.getElementById('st-back-btn');
  var backTitle = document.getElementById('st-back-title');
  var panels  = document.querySelectorAll('.st-panel');

  var labelMap = {
    'security':    'Security',
    'sessions':    'Sessions',
    'preferences': 'Preferences',
    'api-keys':    'API Keys',
    'referrals':   'Referrals'
  };

  function openPanel(name) {
    panels.forEach(function(p) { p.classList.remove('active'); });
    var target = document.getElementById('panel-' + name);
    if (target) target.classList.add('active');
    if (backTitle) backTitle.textContent = labelMap[name] || name;
    layout.classList.add('st-panel-open');
  }

  function closePanel() {
    layout.classList.remove('st-panel-open');
  }

  document.querySelectorAll('[data-st-tab]').forEach(function(btn) {
    btn.addEventListener('click', function() {
      openPanel(btn.getAttribute('data-st-tab'));
    });
  });

  backBtn.addEventListener('click', closePanel);
</script>
